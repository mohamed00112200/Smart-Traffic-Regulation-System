#include <Servo.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// إعداد السيرفو والـ LCD
Servo myServo;
LiquidCrystal_I2C lcd(0x27, 16, 2); // العنوان الافتراضي لـ I2C ممكن يكون 0x3F حسب النوع

// الحساسات
int gateIR = 2;
int lane1 = 7;
int lane2 = 8;
int lane3 = 9;
int lastState = HIGH;

// الحالة السابقة للمحتوى المعروض
String lastLine1 = "";
String lastLine2 = "";

void setup() {
  myServo.attach(3);
  pinMode(gateIR, INPUT);
  pinMode(lane1, INPUT);
  pinMode(lane2, INPUT);
  pinMode(lane3, INPUT);

  myServo.write(0); // يبدأ مغلق
  lcd.init();
  lcd.backlight();
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("  System Ready  ");
  delay(3000);
  lcd.clear();
}

void loop() {
  int currentState = digitalRead(gateIR);
  int l1 = digitalRead(lane1);
  int l2 = digitalRead(lane2);
  int l3 = digitalRead(lane3);

  // عند امتلاء كل الممرات
  if (l1 == LOW && l2 == LOW && l3 == LOW) {
    updateLCD(" FULL CAPACITY ", "   No Parking   ");

    // غلق السيرفو
    for (int i = myServo.read(); i >= 0; i--) {
      myServo.write(i);
      delay(10);
    }
    lastState = HIGH;
    delay(500);
    return;
  }

  // فتح وغلق السيرفو
  if (currentState != lastState) {
    if (currentState == LOW) {
      // طباعة رسالة الفتح
      updateLCD(" Opening Gate ", " Please Wait... ");
      for (int i = myServo.read(); i <= 70; i++) {
        myServo.write(i);
        delay(10);
      }
      delay(2000); // تظهر الرسالة لمدة 2 ثانية
    } else {
      delay(2000); // تأخير قبل الإغلاق
      for (int i = myServo.read(); i >= 0; i--) {
        myServo.write(i);
        delay(10);
      }
    }
    lastState = currentState;
  }

  // عرض الحالة بشكل مستقر بعد الفتح أو في الوضع الطبيعي
  String line1 = "L1:" + String(l1 == LOW ? "Car" : "Emp") +
                 " L2:" + String(l2 == LOW ? "Car" : "Emp");
  String line2 = "L3:" + String(l3 == LOW ? "Car" : "Emp");

  updateLCD(line1, line2);

  delay(200);
}

// تحديث الشاشة فقط إذا تغيّر المحتوى
void updateLCD(String line1, String line2) {
  if (line1 != lastLine1) {
    lcd.setCursor(0, 0);
    lcd.print("                "); // مسح السطر
    lcd.setCursor(0, 0);
    lcd.print(line1);
    lastLine1 = line1;
  }

  if (line2 != lastLine2) {
    lcd.setCursor(0, 1);
    lcd.print("                "); // مسح السطر
    lcd.setCursor(0, 1);
    lcd.print(line2);
    lastLine2 = line2;
  }
}
